cmake_minimum_required(VERSION 3.16)
project(cargo_robot_description)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(CargoHandling SHARED src/CargoHandling.cc)
target_link_libraries(CargoHandling PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-plugin2::register)
install(TARGETS CargoHandling LIBRARY DESTINATION lib)
install(DIRECTORY models urdf worlds launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/demo_cycle.py scripts/manual_control.py scripts/urdf_scissor_demo.py scripts/battery_estimator.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/packing_plan.py scripts/battery_model.py DESTINATION lib/${PROJECT_NAME})
ament_package()
