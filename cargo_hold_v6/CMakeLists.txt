cmake_minimum_required(VERSION 3.16)
project(cargo_hold_v6)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(cppzmq REQUIRED)
if(TARGET cppzmq AND NOT TARGET CPPZMQ::CPPZMQ)
  add_library(CPPZMQ::CPPZMQ ALIAS cppzmq)
endif()
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(ScenarioReplay SHARED src/ScenarioReplay.cc)
target_link_libraries(ScenarioReplay PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
# Also compile the optional v5 force-based handling system; its full dynamic
# ROS 2 integration is a separate research mode, not the cinematic replay.
add_library(CargoHandling SHARED ros2_ws/src/cargo_robot_description/src/CargoHandling.cc)
target_link_libraries(CargoHandling PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
add_executable(replay_ecm_check src/ReplayEcmCheck.cc)
target_link_libraries(replay_ecm_check PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
enable_testing()
add_test(NAME replay_core COMMAND replay_ecm_check)
set_tests_properties(replay_core PROPERTIES ENVIRONMENT "CARGO_PROJECT_ROOT=${CMAKE_CURRENT_SOURCE_DIR};CARGO_REPLAY_SPEED=1;CARGO_REPLAY_RECEIPT=${CMAKE_CURRENT_SOURCE_DIR}/validation/plugin_ecm_execution.json")
